#### 撮合算法

```typescript
class MatchingEngine {
  private orderBooks: Map<string, OrderBook>;

  // 撮合订单
  async matchOrder(order: Order): Promise<MatchResult> {
    const orderBook = this.orderBooks.get(order.symbol);
    const matches: Trade[] = [];

    if (order.side === 'BUY') {
      // 从最优卖价开始撮合
      while (order.quantity > order.filledQuantity) {
        const bestAsk = orderBook.getBestAsk();
        if (!bestAsk || order.price < bestAsk) break;

        // 执行撮合
        const match = this.executeTrade(order, bestAsk);
        matches.push(match);
      }
    } else {
      // 从最优买价开始撮合
      while (order.quantity > order.filledQuantity) {
        const bestBid = orderBook.getBestBid();
        if (!bestBid || order.price > bestBid) break;

        // 执行撮合
        const match = this.executeTrade(order, bestBid);
        matches.push(match);
      }
    }

    return { matches, remainingOrder: order };
  }

  // 执行成交
  private executeTrade(takerOrder: Order, makerOrder: Order): Trade {
    const quantity = Math.min(
      takerOrder.quantity - takerOrder.filledQuantity,
      makerOrder.quantity - makerOrder.filledQuantity
    );

    const price = makerOrder.price; // 使用 Maker 价格

    // 更新订单状态
    takerOrder.filledQuantity += quantity;
    makerOrder.filledQuantity += quantity;

    // 创建成交记录
    const trade = {
      tradeId: generateTradeId(),
      symbol: takerOrder.symbol,
      price,
      quantity,
      takerOrder,
      makerOrder,
      timestamp: Date.now()
    };

    return trade;
  }
}
```

### 3.2 风控引擎设计

```typescript
class RiskEngine {
  // 检查订单风险
  async checkOrderRisk(order: Order, user: User): Promise<RiskCheckResult> {
    const checks = [
      this.checkAccountBalance(order, user),
      this.checkPositionLimit(order, user),
      this.checkOrderLimit(order, user),
      this.checkPriceDeviation(order),
      this.checkRateLimit(user)
    ];

    const results = await Promise.all(checks);
    const passed = results.every(r => r.passed);

    return {
      passed,
      reasons: results.filter(r => !r.passed).map(r => r.reason)
    };
  }

  // 检查账户余额
  private async checkAccountBalance(order: Order, user: User): Promise<CheckResult> {
    const requiredMargin = this.calculateRequiredMargin(order);
    const availableBalance = await this.getAvailableBalance(user.id);

    return {
      passed: availableBalance >= requiredMargin,
      reason: '可用余额不足'
    };
  }

  // 计算所需保证金
  private calculateRequiredMargin(order: Order): Decimal {
    const notional = order.price * order.quantity;
    const leverage = order.leverage || 1;
    const marginRate = 1 / leverage;

    return notional * marginRate;
  }

  // 监控清算风险
  async monitorLiquidationRisk(): Promise<void> {
    const positions = await this.getAllPositions();

    for (const position of positions) {
      const markPrice = await this.getMarkPrice(position.symbol);
      const marginRatio = this.calculateMarginRatio(position, markPrice);

      // 触发清算
      if (marginRatio <= position.maintenanceMarginRate) {
        await this.liquidatePosition(position, markPrice);
      }
      // 发送预警
      else if (marginRatio <= position.maintenanceMarginRate * 1.2) {
        await this.sendRiskWarning(position.userId, position);
      }
    }
  }

  // 计算保证金率
  private calculateMarginRatio(position: Position, markPrice: Decimal): Decimal {
    const positionValue = position.quantity * markPrice;
    const unrealizedPnl = this.calculateUnrealizedPnl(position, markPrice);
    const equity = position.margin + unrealizedPnl;

    return equity / positionValue;
  }
}
```

```typescript
  // 清算持仓
  async liquidatePosition(position: Position, markPrice: Decimal): Promise<void> {
    // 1. 创建强平订单
    const liquidationOrder: Order = {
      orderId: generateOrderId(),
      userId: 'LIQUIDATION_BOT',
      symbol: position.symbol,
      side: position.side === 'LONG' ? 'SELL' : 'BUY',
      type: 'MARKET',
      quantity: position.quantity,
      reduceOnly: true,
      timestamp: Date.now()
    };

    // 2. 提交订单到撮合引擎
    const matchResult = await this.matchingEngine.matchOrder(liquidationOrder);

    // 3. 计算保证金损失
    const marginLoss = this.calculateMarginLoss(position, matchResult);

    // 4. 更新保险基金
    await this.updateInsuranceFund(marginLoss);

    // 5. 记录清算记录
    await this.recordLiquidation({
      userId: position.userId,
      symbol: position.symbol,
      side: position.side,
      quantity: position.quantity,
      price: matchResult.averagePrice,
      marginLoss,
      timestamp: Date.now()
    });

    // 6. 发送通知
    await this.notifyUser(position.userId, {
      type: 'LIQUIDATION',
      position,
      price: matchResult.averagePrice
    });
  }

  // ADL (自动减仓)
  async executeADL(symbol: string, side: 'LONG' | 'SHORT'): Promise<void> {
    // 1. 获取同方向盈利最高的持仓
    const positions = await this.getTopProfitablePositions(symbol, side, 10);

    // 2. 按盈利率排序
    const sortedPositions = positions.sort((a, b) =>
      b.unrealizedPnlRate - a.unrealizedPnlRate
    );

    // 3. 依次减仓
    for (const position of sortedPositions) {
      await this.reducePosition(position, position.quantity * 0.5);
    }
  }
}
```

```typescript
  // 验证签名
  async verifySignature(
    message: string,
    signature: string,
    address: string
  ): Promise<boolean> {
    try {
      const recoveredAddress = ethers.verifyMessage(message, signature);
      return recoveredAddress.toLowerCase() === address.toLowerCase();
    } catch (error) {
      return false;
    }
  }

  // 验证订单签名
  async verifyOrderSignature(order: Order, signature: string): Promise<boolean> {
    // 构造 EIP-712 消息
    const domain = {
      name: 'Hyperliquid',
      version: '1',
      chainId: 1,
      verifyingContract: '0x...'
    };

    const types = {
      Order: [
        { name: 'symbol', type: 'string' },
        { name: 'side', type: 'string' },
        { name: 'price', type: 'uint256' },
        { name: 'quantity', type: 'uint256' },
        { name: 'nonce', type: 'uint256' },
        { name: 'expiry', type: 'uint256' }
      ]
    };

    const value = {
      symbol: order.symbol,
      side: order.side,
      price: order.price.toString(),
      quantity: order.quantity.toString(),
      nonce: order.nonce,
      expiry: order.expiry
    };

    // 验证签名
    const digest = ethers.TypedDataEncoder.hash(domain, types, value);
    const recoveredAddress = ethers.recoverAddress(digest, signature);

    return recoveredAddress.toLowerCase() === order.userAddress.toLowerCase();
  }

  // 生成 API Key
  async generateApiKey(userId: number): Promise<{ apiKey: string, apiSecret: string }> {
    const apiKey = crypto.randomBytes(32).toString('hex');
    const apiSecret = crypto.randomBytes(64).toString('hex');

    // 加密存储 secret
    const encryptedSecret = await bcrypt.hash(apiSecret, 10);

    await this.db.apiKeys.create({
      userId,
      apiKey,
      apiSecret: encryptedSecret,
      permissions: ['trade', 'read'],
      status: 1
    });

    return { apiKey, apiSecret };
  }
```

### 5.4 WebSocket 管理

```typescript
// composables/useWebSocket.ts
import { ref, onMounted, onUnmounted } from 'vue';

export function useWebSocket(url: string) {
  const ws = ref<WebSocket | null>(null);
  const connected = ref(false);
  const messageQueue = ref<any[]>([]);

  const connect = () => {
    ws.value = new WebSocket(url);

    ws.value.onopen = () => {
      connected.value = true;
      console.log('WebSocket connected');

      // 发送队列中的消息
      while (messageQueue.value.length > 0) {
        const msg = messageQueue.value.shift();
        send(msg);
      }
    };

    ws.value.onclose = () => {
      connected.value = false;
      console.log('WebSocket disconnected');

      // 重连
      setTimeout(connect, 3000);
    };

    ws.value.onerror = (error) => {
      console.error('WebSocket error:', error);
    };
  };

  const send = (data: any) => {
    if (connected.value && ws.value) {
      ws.value.send(JSON.stringify(data));
    } else {
      messageQueue.value.push(data);
    }
  };

  const subscribe = (channel: string, params?: any) => {
    send({
      method: 'subscribe',
      params: { channel, ...params },
      id: Date.now()
    });
  };

  const unsubscribe = (channel: string, params?: any) => {
    send({
      method: 'unsubscribe',
      params: { channel, ...params },
      id: Date.now()
    });
  };

  const on = (handler: (data: any) => void) => {
    if (ws.value) {
      ws.value.onmessage = (event) => {
        const data = JSON.parse(event.data);
        handler(data);
      };
    }
  };

  onMounted(() => {
    connect();
  });

  onUnmounted(() => {
    if (ws.value) {
      ws.value.close();
    }
  });

  return {
    connected,
    send,
    subscribe,
    unsubscribe,
    on
  };
}
```

```typescript
// logger.ts
import winston from 'winston';
import path from 'path';

const logFormat = winston.format.combine(
  winston.format.timestamp({ format: 'YYYY-MM-DD HH:mm:ss' }),
  winston.format.errors({ stack: true }),
  winston.format.splat(),
  winston.format.json()
);

const logger = winston.createLogger({
  level: process.env.LOG_LEVEL || 'info',
  format: logFormat,
  transports: [
    // 错误日志
    new winston.transports.File({
      filename: path.join('logs', 'error.log'),
      level: 'error',
      maxsize: 10485760, // 10MB
      maxFiles: 5
    }),
    // 综合日志
    new winston.transports.File({
      filename: path.join('logs', 'combined.log'),
      maxsize: 10485760,
      maxFiles: 5
    }),
    // 控制台输出
    new winston.transports.Console({
      format: winston.format.combine(
        winston.format.colorize(),
        winston.format.simple()
      )
    })
  ]
});

export default logger;
```
